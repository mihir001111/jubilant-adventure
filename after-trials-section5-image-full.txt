<script id="pillars-always-active-script">
(function () {
  function initPillars() {
    const section = document.querySelector('.pillars-section');
    if (!section) return;

    const panels = Array.from(section.querySelectorAll('.pillar-panel'));
    if (!panels.length) return;

    let activePanel = panels[1] || panels[0];

    function setActive(panel) {
      if (!panels.includes(panel)) return;
      activePanel = panel;
      panels.forEach(function (item) {
        const isActive = item === activePanel;
        item.classList.toggle('active', isActive);
        item.setAttribute('aria-selected', isActive ? 'true' : 'false');
      });
    }

    /* Panel 2 is active immediately, so its background and content are visible on load. */
    setActive(activePanel);

    panels.forEach(function (panel) {
      panel.setAttribute('role', 'tab');
      panel.setAttribute('tabindex', '0');

      panel.addEventListener('mouseenter', function () {
        setActive(panel);
      });

      panel.addEventListener('focusin', function () {
        setActive(panel);
      });

      panel.addEventListener('click', function () {
        setActive(panel);
      });

      panel.addEventListener('keydown', function (event) {
        if (event.key === 'Enter' || event.key === ' ') {
          event.preventDefault();
          setActive(panel);
        }
      });
    });

    /* On desktop, return to the first tab when leaving the whole group. */
    section.querySelector('.pillars-panels')?.addEventListener('mouseleave', function () {
      setActive(activePanel || panels[1] || panels[0]);
    });
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initPillars);
  } else {
    initPillars();
  }
})();
</script>
